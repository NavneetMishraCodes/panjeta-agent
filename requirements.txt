# Panjeta Agent - project dependencies
#
# Intentionally minimal: no AI frameworks (e.g. LangChain, LangGraph),
# no vector databases, no memory systems. Dependencies are introduced
# only as they become necessary.

# Official OpenAI SDK - used because OpenRouter exposes an
# OpenAI-compatible API. This is NOT an OpenAI-the-company integration.
openai

# Official Google Gemini SDK - used by the "google" provider adapter.
google-genai

# Loads environment variables (e.g. OPENROUTER_API_KEY, GOOGLE_API_KEY)
# from a local .env file for development.
python-dotenv
